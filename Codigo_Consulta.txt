      // >>> TAREA: CONSULTA DE USUARIOS Y PUBLICACIONES (FETCH) <<<
      // API: JSONPlaceholder -> /users  y  /posts?userId=ID

      const API_URL = 'https://jsonplaceholder.typicode.com';
      const EMOJIS_USUARIO = ['🧑‍🎨', '👩‍🚀', '🧑‍🍳', '👨‍🔬', '👩‍💻', '🧑‍🏫', '👨‍🌾', '👩‍⚕️', '🧑‍🚒', '👨‍🎤'];

      const cqEnlace = document.getElementById('enlace-consulta');
      const cqLista = document.getElementById('cq-lista');
      const cqBuscar = document.getElementById('cq-buscar');
      const cqContador = document.getElementById('cq-contador');
      const cqDetalle = document.getElementById('cq-detalle');

      let cqUsuarios = [];
      let cqTexto = '';
      let cqSeleccionado = null;
      let cqCargado = false;
      let cqCargando = false;
      let cqControlador = null; // AbortController de la consulta de publicaciones en curso

      const normalizar = (s) => String(s).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');

      const cqAvatar = (u, grande) =>
        `<span class="cq-avatar cq-c${(u.id - 1) % 5}${grande ? ' cq-avatar--grande' : ''}" aria-hidden="true">${EMOJIS_USUARIO[(u.id - 1) % EMOJIS_USUARIO.length]}</span>`;

      const cqRenderLista = () => {
        const t = normalizar(cqTexto);
        // Si lo escrito es solo un número (ej. "3" o "#3"), se busca por número de usuario (id)
        const porNumero = /^#?\d+$/.test(cqTexto);
        const idBuscado = porNumero ? Number(cqTexto.replace('#', '')) : null;
        const filtrados = cqUsuarios.filter(u => {
          if (!t) return true;
          if (porNumero) return u.id === idBuscado;
          return normalizar([u.name, u.username, u.email, u.company?.name ?? ''].join(' ')).includes(t);
        });

        cqContador.textContent = `👥 Mostrando ${filtrados.length} de ${cqUsuarios.length} usuarios`;

        if (filtrados.length === 0) {
          cqLista.innerHTML = porNumero
            ? `<div class="cq-vacio">🔎 No existe el usuario #${idBuscado}. Prueba con un número del 1 al ${cqUsuarios.length}.</div>`
            : `<div class="cq-vacio">🔎 No hay usuarios que coincidan con “${escaparHTML(cqTexto)}”.</div>`;
          return;
        }

        cqLista.innerHTML = filtrados.map(u => `
          <button type="button" class="cq-usuario${u.id === cqSeleccionado ? ' activo' : ''}" data-id="${u.id}" aria-pressed="${u.id === cqSeleccionado}">
            ${cqAvatar(u)}
            <span class="cq-usuario-info">
              <span class="cq-usuario-nombre"><span class="cq-numero">#${u.id}</span>${escaparHTML(u.name)}</span>
              <span class="cq-usuario-dato">✉️ ${escaparHTML(u.email)}</span>
              <span class="cq-usuario-dato">🏢 ${escaparHTML(u.company?.name ?? '—')}</span>
            </span>
          </button>`).join('');
      };

      // --------------------------------------------------------------
      // >>> CONSULTA 1: GET /users  (lista de usuarios)
    
      const cqCargarUsuarios = async () => {
        if (cqCargado || cqCargando) return;
        cqCargando = true;
        cqContador.textContent = '';
        cqLista.innerHTML = '<div class="cq-estado">⏳ Cargando usuarios...</div>';
        try {
          const res = await fetch(`${API_URL}/users`);
          if (!res.ok) throw new Error('HTTP ' + res.status);
          const data = await res.json();
          if (!Array.isArray(data)) throw new Error('Formato inválido');
          cqUsuarios = data;
          cqCargado = true;
          cqRenderLista();
        } catch (err) {
          console.error('Error cargando usuarios:', err);
          cqLista.innerHTML = `<div class="cq-estado cq-estado--error">⚠️ No se pudieron cargar los usuarios. Revisa tu conexión e inténtalo de nuevo.<br><button type="button" class="btn btn-dorado" id="cq-reintentar-usuarios">Reintentar</button></div>`;
        } finally {
          cqCargando = false;
        }
      };

      const cqCabeceraUsuario = (u) => `
        <div class="cq-perfil">
          ${cqAvatar(u, true)}
          <div class="cq-perfil-info">
            <div class="cq-perfil-nombre">${escaparHTML(u.name)}</div>
            <div class="cq-perfil-user">Usuario #${u.id} · @${escaparHTML(u.username)}</div>
            <div class="cq-chips">
              <span class="cq-chip">✉️ ${escaparHTML(u.email)}</span>
              <span class="cq-chip">🏢 ${escaparHTML(u.company?.name ?? '—')}</span>
              <span class="cq-chip">📞 ${escaparHTML(u.phone ?? '—')}</span>
              <span class="cq-chip">🌐 ${escaparHTML(u.website ?? '—')}</span>
              <span class="cq-chip">📍 ${escaparHTML(u.address?.city ?? '—')}</span>
            </div>
            <div class="cq-frase">“${escaparHTML(u.company?.catchPhrase ?? '')}”</div>
          </div>
        </div>`;

      // --------------------------------------------------------------
      // >>> CONSULTA 2: GET /posts?userId=ID  (publicaciones del usuario)
      // El ID se reemplaza por el del usuario seleccionado.
     
      const cqCargarPosts = async (id) => {
        if (cqControlador) cqControlador.abort(); // cancela la consulta anterior si cambian de usuario
        cqControlador = new AbortController();
        const { signal } = cqControlador;
        const zona = document.getElementById('cq-posts');
        zona.innerHTML = '<div class="cq-estado">⏳ Cargando publicaciones...</div>';

        try {
          const res = await fetch(`${API_URL}/posts?userId=${encodeURIComponent(id)}`, { signal });
          if (!res.ok) throw new Error('HTTP ' + res.status);
          const posts = await res.json();
          if (!Array.isArray(posts)) throw new Error('Formato inválido');

          if (posts.length === 0) {
            zona.innerHTML = '<div class="cq-vacio">📭 Este usuario aún no tiene publicaciones.</div>';
            return;
          }
          zona.innerHTML = `<div class="cq-posts-titulo">📝 Publicaciones <span class="cq-badge">${posts.length}</span></div>` +
            posts.map(p => `
              <article class="cq-post">
                <h4 class="cq-post-titulo">📌 ${escaparHTML(p.title)}</h4>
                <div class="cq-post-cuerpo">${escaparHTML(p.body)}</div>
              </article>`).join('');
        } catch (err) {
          if (err.name === 'AbortError') return; // fue cancelada a propósito, no es un error
          console.error('Error cargando publicaciones:', err);
          zona.innerHTML = `<div class="cq-estado cq-estado--error">⚠️ No se pudieron cargar las publicaciones. Revisa tu conexión e inténtalo de nuevo.<br><button type="button" class="btn btn-dorado" data-reintentar-posts="${id}">Reintentar</button></div>`;
        }
      };

      const cqSeleccionar = (id, desdeBusqueda) => {
        const u = cqUsuarios.find(x => x.id === id);
        if (!u) return;
        cqSeleccionado = id;

        cqLista.querySelectorAll('.cq-usuario').forEach(b => {
          const activo = Number(b.dataset.id) === id;
          b.classList.toggle('activo', activo);
          b.setAttribute('aria-pressed', String(activo));
        });

        cqDetalle.innerHTML = cqCabeceraUsuario(u) + '<div class="cq-posts" id="cq-posts"></div>';
        cqDetalle.scrollTop = 0;
        if (!desdeBusqueda && window.matchMedia('(max-width: 760px)').matches) {
          cqDetalle.scrollIntoView({ behavior: reduceMotion ? 'auto' : 'smooth', block: 'start' });
        }
        cqCargarPosts(id);
      };

      // Eventos de la consulta
      cqEnlace.addEventListener('click', (e) => {
        e.preventDefault(); // el modal lo abre el sistema data-abrir; aquí solo cargamos los datos
        cqCargarUsuarios();
      });

      cqLista.addEventListener('click', (e) => {
        if (e.target.closest('#cq-reintentar-usuarios')) { cqCargarUsuarios(); return; }
        const tarjeta = e.target.closest('.cq-usuario');
        if (tarjeta) cqSeleccionar(Number(tarjeta.dataset.id));
      });

      cqDetalle.addEventListener('click', (e) => {
        const reintentar = e.target.closest('[data-reintentar-posts]');
        if (reintentar) cqCargarPosts(Number(reintentar.dataset.reintentarPosts));
      });

      cqBuscar.addEventListener('input', () => {
        cqTexto = cqBuscar.value.trim();
        if (!cqCargado) return;
        cqRenderLista();

        // Si escribió un número válido, lo lleva directo a ese usuario
        if (/^#?\d+$/.test(cqTexto)) {
          const id = Number(cqTexto.replace('#', ''));
          if (cqUsuarios.some(u => u.id === id) && id !== cqSeleccionado) cqSeleccionar(id, true);
        }
      });
